```

auto lo
iface lo inet loopback
auto ens33
iface ens33 inet static
    address 192.168.1.10
    netmask 255.255.255.0
    gateway 192.168.1.1

EOF

sudo systemctl restart networking
```

<summary>Linux 挂载硬盘</summary> 
> 适用于 Debian / Ubuntu / Alpine
<a name="10"></a>
<hr style="all: unset; display: block; margin: 12px auto; height: 6px; border-top: 1px solid #7ee7878f; width: 100%;">
<!-- ====================================================================== -->
```
lsblk	#查看有哪些硬盘或分区
fdisk -l #详细
df -h 
```
```
mkdir /mnt/data 	#创建挂载目录
mount /dev/sdb1 /mnt/data 	#挂载到目录
df -h 	#验证查看
ls /mnt/data 	#查看目录
```
===========
 
```
开机自动挂载
blkid /dev/sdb1	#查看分区 UUID

nano /etc/fstab 	#编辑 fstab 	
UUID=你复制的UUID   /mnt/data   ext4   defaults   0  2

mount -a 	#测试是否正确
```
```
apt update
apt install -y ntfs-3g #安装这个驱动就可以查看 windows 的硬盘内容
```

<details>  
<summary>Certbot</summary> 
> 适用于 Debian / Ubuntu / ssh 自动续期证书
<a name="11"></a>
<hr style="all: unset; display: block; margin: 12px auto; height: 6px; border-top: 1px solid #7ee7878f; width: 100%;">
<!-- ====================================================================== -->
> 安装 及插件
 
```
sudo apt update && sudo apt install -y certbot python3-pip

python3 -m venv certbot-venv
source certbot-venv/bin/activate
pip install certbot certbot-dns-cloudflare certbot-dns-aliyun certbot-dns-dnspod zope-interface zope-component \
-i https://pypi.tuna.tsinghua.edu.cn/simple
Cloudflare
apt install -y python3-certbot-dns-cloudflare

阿里云（Aliyun）
apt install -y python3-certbot-dns-aliyun

腾讯云（DNSPod）
apt install -y python3-certbot-dns-dnspod

安装完你可以直接用 来验证插件是否安装成功
certbot plugins
```
> dns泛域名证书申请 示例
```
/app/certbot-dns/certbot-venv/bin/certbot certonly \
  --dns-cloudflare \
  --dns-cloudflare-credentials /app/certbot-dns/cloudflare.ini \
  --dns-cloudflare-propagation-seconds 60 \
  -d sou.onl \
  -d "*.sou.onl" \
  --email 1009721777@qq.com \
  --agree-tos \
  --no-eff-email \
  --force-renewal \
  --deploy-hook "cat /etc/letsencrypt/live/sou.onl/fullchain.pem /etc/letsencrypt/live/sou.onl/privkey.pem | ssh -o StrictHostKeyChecking=no root@47.74.7.172 'cat > /etc/haproxy/ssl/sou.onl.pem && systemctl restart haproxy' && cp /etc/letsencrypt/live/sou.onl/*.pem /app/frp/ && docker compose -f /app/frp/docker-compose.yml up -d --force-recreate"
  ```
> 手动续签
```
/app/certbot-dns/bin/certbot renew --dns-cloudflare-propagation-seconds 60 --force-renewal #手动强制续签
```
> 查看证书状态
```
certbot certificates #查询证书状态
```
> 删除证书
```
/app/certbot-dns/certbot-venv/bin/certbot delete --cert-name book.sou.onl
```
